\section{1. Problem Statement}
\begin{itemize}
    \item THz (300\,GHz) signals suffer from severe path loss, molecular absorption, and blockage.
    \item Fixed access points cannot adapt to changing user/crowd distributions.
    \item Electronic beamforming alone fails when all line-of-sight paths are blocked by walls or human bodies.
    \item Baseline coverage drops to 60--75\% in dynamic indoor environments---far below the 90\% target.
\end{itemize}

\section{2. Proposed Solution}
\begin{itemize}
    \item 14 THz routers mounted on \textbf{motorized ceiling rails} for physical repositioning.
    \item \textbf{APSO (Adaptive Particle Swarm Optimization)} computes optimal router positions in real-time.
    \item \textbf{4-Tier Escalation Hierarchy} to minimize unnecessary mechanical movement:
    \begin{itemize}
        \item Tier 1--2: Electronic adaptation (beam azimuth $\phi$, beamwidth $\alpha$).
        \item Tier 3: Vertical height adjustment ($Z$-axis, 2.0\,m--3.2\,m).
        \item Tier 4: Full 3D spatial repositioning ($X, Y, Z$).
    \end{itemize}
    \item \textbf{Hungarian Algorithm} for minimum-displacement router-to-target assignment.
\end{itemize}

\section{3. Simulation Setup}
\begin{itemize}
    \item \textbf{Room}: $32\,\text{m} \times 20\,\text{m} \times 3\,\text{m}$ with realistic wall layout (concrete, wood, glass).
    \item \textbf{Routers}: 14 THz APs at 300\,GHz, 20\,dBm Tx power, 30\,dBi max antenna gain.
    \item \textbf{Channel Model}: FSPL + molecular absorption ($k_f = 0.05$) + 3D ray-wall intersection + human body blockage (20\,dB).
    \item \textbf{Material Losses}: Concrete 30\,dB, Wood 20\,dB, Glass 15\,dB.
    \item \textbf{APSO Config}: 200 particles, 70 dimensions ($14 \times 5$ params), 200 iterations, adaptive inertia ($0.9 \to 0.4$).
    \item \textbf{Fitness Function}: Density-weighted coverage (0.6) + cell coverage (0.4) + target bonus $-$ movement penalty $-$ anti-clustering penalty.
    \item \textbf{Time Steps}: 3 dynamic crowd scenarios ($t_0, t_1, t_2$) with shifting Gaussian user distributions.
\end{itemize}

\section{4. Results \& Evidence}
\begin{itemize}
    \item \textbf{Coverage Improvement}:
    \begin{itemize}
        \item Before APSO: $\sim$60--75\%.
        \item After APSO: $\geq$90\% (target met across all 3 time steps).
    \end{itemize}
    \item \textbf{Optimization Speed}: $<$2.5 seconds per cycle (real-time feasible).
    \item \textbf{Total Router Displacement}: $<$5--12 meters combined across all 14 routers.
    \item \textbf{Reachability Gain}: +15--35\% coverage increase per meter of movement.
    \item \textbf{Dead Zone Reduction}: $\sim$90\% elimination of poor-SINR areas.
    \item \textbf{SINR Heatmaps}: Clear visual improvement---dead zones before $\to$ uniform coverage after.
    \item Coverage maintained consistently across all dynamic crowd shifts.
\end{itemize}
